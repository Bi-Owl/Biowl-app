server {
    listen 80;
    server_name biowl.ir;

    # Set maximum allowed size of the client request body.
    # This needs to be at least 15M to accommodate image and PDF uploads.
    client_max_body_size 20M; # Set to 20MB to be safe, slightly above 15MB

    # Root directory for the frontend build files
    root /var/www/biowl-app/frontend/dist;
    index index.html;

    # Serve static assets and handle SPA routing
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Proxy API requests to the backend server
    location /api/ {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    # Serve uploaded files
    location /uploads/ {
        alias /var/www/biowl-app/backend/uploads/;
        try_files $uri =404;
    }
}
